const response = await fetch('/api/v1/runs?simulationId=SIM_001');
if (!response.ok) throw new Error(`查询失败：HTTP ${response.status}`);

const runs = await response.json();
const active = runs.find(r => r.ended_at_ms == null);

if (active) {
  const result = await fetch(`/api/v1/runs/${active.run_id}/finish`, {
    method: 'POST',
    headers: {'Content-Type': 'application/json'},
    body: '{}'
  });

  console.log('结束状态：', result.status);
  console.log(await result.json());
} else {
  console.log('SIM_001 没有未结束记录');
}
